fn print_number() {
  %t0 = load %p0
  %t1 = add %t0, 100
  call print, %t1
  %t2 = load %p1
  %t3 = add %t2, 101
  call print, %t3
}

fn main() {
  %t4 = create int
  store 100, %t4
  %t5 = create int
  store 200, %t5
  %t6 = create int
  %t7 = load %t4
  %t8 = load %t5
  %t9 = add %t7, %t8
  %t10 = mul %t9, 2
  %t11 = load %t4
  %t12 = mul 3, %t11
  %t13 = add %t10, %t12
  store %t13, %t6
  store 50, %p0
  %t14 = call print_number
  store 80, %p0
  %t15 = call print_number
  %t16 = load %t6
  %t17 = eq %t16, 900
  cond_jump %t17, then1
  jump else2
  label then1
  call print, 111
  %t18 = add 111, 333
  call print, %t18
  jump endif0
  label else2
  call print, 555
  jump endif0
  label endif0
  store 500, %t5
  store 600, %t4
  %t19 = load %t4
  %t20 = load %t5
  %t21 = gt %t19, %t20
  cond_jump %t21, then4
  jump endif3
  label then4
  call print, 777
  jump endif3
  label endif3
  %t22 = load %t4
  %t23 = load %t5
  %t24 = lt %t22, %t23
  cond_jump %t24, then6
  jump endif5
  label then6
  call print, 77777777
  jump endif5
  label endif5
  %t25 = load %t4
  %t26 = eq %t25, 600
  cond_jump %t26, then8
  jump endif7
  label then8
  call print, 888
  jump endif7
  label endif7
  %t27 = load %t5
  %t28 = lt %t27, 100
  cond_jump %t28, then10
  jump endif9
  label then10
  call print, 999
  jump endif9
  label endif9
}
